docker build -f k3s/sky/Dockerfile -t takenking9879/sky-runner:0.12.0 .


All Airflow sky-runner KPO pods become lightweight HTTP clients via
SKYPILOT_API_SERVER_ENDPOINT (added to env-secret). Zero code changes to DAGs or
sky_runner.py -- the SkyPilot SDK picks up the env var automatically.


PREREQUISITES
-------------
- kubectl configured for the target cluster
- helm 3.x installed (helm version to verify)
- apache2-utils installed for htpasswd (sudo apt install apache2-utils)
- .env file in repo root with: AWS_ACCESS_KEY_ID, AWS_SECRET_ACCESS_KEY,
  AWS_DEFAULT_REGION, RUNPOD_API_KEY, VASTAI_API_KEY, MLFLOW_TRACKING_URI
- An S3 bucket to use for SkyPilot workdir persistence (e.g. s3://my-sky-workdir)


helm repo add skypilot https://helm.skypilot.co/

kubectl create namespace skypilot

kubectl create secret generic runpod-credentials \
  -n skypilot \
  --from-literal=api_key=$(grep RUNPOD_API_KEY .env | cut -d '=' -f2)

kubectl create secret generic vastai-credentials \
  -n skypilot \
  --from-literal=api_key=$(grep -E 'VASTAI_API_KEY|VAST_API_KEY' .env | head -n1 | cut -d '=' -f2)

kubectl create secret generic aws-credentials \
  -n skypilot \
  --from-literal=aws_access_key_id=$(grep AWS_ACCESS_KEY_ID .env | cut -d '=' -f2) \
  --from-literal=aws_secret_access_key=$(grep AWS_SECRET_ACCESS_KEY .env | cut -d '=' -f2) \
  --from-literal=aws_default_region=$(grep AWS_DEFAULT_REGION .env | cut -d '=' -f2 || echo us-east-1)

STEP 6: Verify the API server started correctly
------------------------------------------------
kubectl get pods -n skypilot
kubectl logs -n skypilot deployment/my-skypilot-api-server --tail=60

# Expected log lines (from docker-entrypoint.sh):
#   [entrypoint] RunPod credentials written to ~/.runpod/config.toml
#   [entrypoint] VastAI credentials written to ~/.config/vastai/vast_api_key
#   [entrypoint] SkyPilot jobs controller disk_size set to 30 GB
# Then the sky api start output.

# Health check via port-forward:
kubectl port-forward svc/my-skypilot-api-server -n skypilot 46580:46580 &
PF_PID=$!
sleep 2
curl -s http://localhost:46580/api/health
kill $PF_PID
# Expected: {"status":"ok"} or similar



STEP 8: Verify consolidation mode is active (no cloud controller VM)
----------------------------------------------------------------------
kubectl exec -n skypilot deployment/my-skypilot-api-server -- \
  sky status --all 2>&1 | grep -i controller || echo "No controller clusters (consolidation mode OK)"

# Should show NO "jobs-controller-*" cloud cluster entries.
# The controller runs in-process; no RunPod/AWS VM is provisioned for it.
